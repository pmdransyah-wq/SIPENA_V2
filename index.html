const GAS_URL = "https://script.google.com/macros/s/AKfycbwwYDMUrIz8kpr0hpC-czdqDHxC1xhNuMGItgfLVvc8DuzVydnTreQDA6Av18C_4p82/exec"; // ganti dengan URL deploy terbaru function handleLogin(e) { e.preventDefault(); const username = document.getElementById("username").value.trim(); const password = document.getElementById("password").value.trim(); if (!username || !password) { Swal.fire("Peringatan", "Username dan Password wajib diisi", "warning"); return; } Swal.fire({ title: "Memeriksa Akun...", allowOutsideClick: false, didOpen: () => Swal.showLoading() }); fetch(GAS_URL, { method: "POST", body: JSON.stringify({ username, password }) }) .then(res => res.json()) .then(data => { if (data.status === "success") { // Simpan session localStorage.setItem("arsip_logged_in", "true"); localStorage.setItem("arsip_username", data.data.username); localStorage.setItem("arsip_role", data.data.role); localStorage.setItem("arsip_instansi", data.data.instansi); localStorage.setItem("arsip_token", data.data.token); Swal.fire({ icon: "success", title: "Login Berhasil", text: "Selamat datang, " + data.data.username, timer: 1500, showConfirmButton: false }).then(() => { window.location.href = "dashboard.html"; }); } else { Swal.fire("Gagal", data.message || "Username atau Password salah!", "error"); } }) .catch(err => { Swal.fire("Error", "Gagal terhubung ke server", "error"); }); }
